                    let fighterName = winner === 'chong' ? data.chongName : data.hongName;
                    let sideLabel = winner === 'chong' ? 'ฝ่ายน้ำเงิน (CHONG)' : 'ฝ่ายแดง (HONG)';
                    
                    let bigBox = document.getElementById('display-champion-name-big');
                    bigBox.innerText = fighterName;
                    bigBox.className = `text-4xl md:text-6xl font-black font-cyber py-6 px-8 rounded-2xl border-2 my-2 ${winner === 'chong' ? 'bg-blue-950/90 border-blue-400 text-blue-300 glow-blue' : 'bg-red-950/90 border-red-400 text-red-300 glow-red'}`;
                    
                    document.getElementById('display-winner-desc').innerText = `สุดยอดนักกีฬา ${sideLabel} คว้าชัยชนะการแข่งขันรายการนี้ด้วยสกอร์ ${data.chongRoundsWon} ต่อ ${data.hongRoundsWon} ยกอย่างยิ่งใหญ่!`;
                    document.getElementById('display-tiebreak-reason').innerText = `ชนะการแข่งขันแบบ Best of 3 ยกสมบูรณ์แบบ`;
                    displayWinnerModal.classList.remove('hidden');
                } else {
                    displayWinnerModal.classList.add('hidden');
                }
            });
        }

        function judgePressPoint(fighter, pts) {
            let now = Date.now();
            matchRef.transaction((data) => {
                if (!data || data.isRestMode) return data;
                let vote = data.pendingVote;
                if (!vote || vote.fighter !== fighter || vote.pts !== pts || (now - vote.time > 1000)) {
                    data.pendingVote = { judge: currentRole, fighter: fighter, pts: pts, time: now };
                    return data;
                } else {
                    if (vote.judge === currentRole) return data;
                    if (fighter === 'chong') {
                        data.chongScore += pts;
                        if (pts === 4 || pts === 6) data.chongSpinHead += pts;
                        if (!data.firstScorer) { data.firstScorer = 'chong'; data.chongFirstPoint = true; }
                    } else {
                        data.hongScore += pts;
                        if (pts === 4 || pts === 6) data.hongSpinHead += pts;
                        if (!data.firstScorer) { data.firstScorer = 'hong'; data.hongFirstPoint = true; }
                    }
                    data.pendingVote = null;
                    if (Math.abs(data.chongScore - data.hongScore) >= 15) {
                        setTimeout(() => handlePointGapWin(data), 100);
                    }
                    return data;
                }
            });

            let alertBox = document.getElementById('judge-status-alert');
            alertBox.classList.remove('hidden');
            setTimeout(() => alertBox.classList.add('hidden'), 1000);
        }

        function scorePoint(fighter, pts) {
            matchRef.transaction((data) => {
                if (!data || data.isRestMode) return data;
                if (fighter === 'chong') {
                    data.chongScore += pts;
                    if (pts === 4 || pts === 6) data.chongSpinHead += pts;
                    if (!data.firstScorer) { data.firstScorer = 'chong'; data.chongFirstPoint = true; }
                } else {
                    data.hongScore += pts;
                    if (pts === 4 || pts === 6) data.hongSpinHead += pts;
                    if (!data.firstScorer) { data.firstScorer = 'hong'; data.hongFirstPoint = true; }
                }
                if (Math.abs(data.chongScore - data.hongScore) >= 15) {
                    setTimeout(() => handlePointGapWin(data), 100);
                }
                return data;
            });
        }

        function toggleTimer() {
            matchRef.once('value', (snapshot) => {
                let data = snapshot.val();
                let running = !data.timerRunning;
                matchRef.update({ timerRunning: running });
                if (running) startIntervalTimer();
                else stopIntervalTimer();
            });
        }

        function startIntervalTimer() {
            if (timerInterval) clearInterval(timerInterval);
            timerInterval = setInterval(() => {
                matchRef.once('value', (snapshot) => {
                    let data = snapshot.val();
                    if (!data || !data.timerRunning) return;
                    let time = data.currentTime - 1;
                    if (time > 0) {
                        matchRef.update({ currentTime: time });
                    } else {
                        matchRef.update({ currentTime: 0, timerRunning: false });
                        stopIntervalTimer();
                        if (!data.isRestMode) handleRoundEnd(data);
                        else autoProceedNextRoundAfterRest(data);
                    }
                });
            }, 1000);
        }

        function stopIntervalTimer() { if (timerInterval) clearInterval(timerInterval); }

        function resetTimer() {
            matchRef.once('value', (snapshot) => {
                let data = snapshot.val();
                let time = data.isRestMode ? data.restTimeSetting : data.matchTimeSetting;
                matchRef.update({ currentTime: time, timerRunning: false });
                stopIntervalTimer();
            });
        }

        function switchTimerMode() {
            matchRef.once('value', (snapshot) => {
                let data = snapshot.val();
                let isRest = !data.isRestMode;
                let time = isRest ? data.restTimeSetting : data.matchTimeSetting;
                matchRef.update({ isRestMode: isRest, currentTime: time, timerRunning: false });
                stopIntervalTimer();
            });
        }

        function endRoundManually() {
            matchRef.once('value', (snapshot) => {
                let data = snapshot.val();
                matchRef.update({ timerRunning: false });
                stopIntervalTimer();
                handleRoundEnd(data);
            });
        }

        function addGamJeom(fighter, penaltyPts) {
            matchRef.transaction((data) => {
                if (!data) return data;
                if (fighter === 'chong') { data.chongGamJeom += 1; data.hongScore += penaltyPts; data.chongLastPenaltyPts = penaltyPts; }
                else { data.hongGamJeom += 1; data.chongScore += penaltyPts; data.hongLastPenaltyPts = penaltyPts; }
                return data;
            });
        }

        function removeGamJeom(fighter) {
            matchRef.transaction((data) => {
                if (!data) return data;
                if (fighter === 'chong' && data.chongGamJeom > 0) {
                    data.chongGamJeom -= 1;
                    data.hongScore = Math.max(0, data.hongScore - (data.chongLastPenaltyPts || 1));
                } else if (fighter === 'hong' && data.hongGamJeom > 0) {
                    data.hongGamJeom -= 1;
                    data.chongScore = Math.max(0, data.chongScore - (data.hongLastPenaltyPts || 1));
                }
                return data;
            });
        }

        function adjustScore(fighter, amount) {
            matchRef.transaction((data) => {
                if (!data) return data;
                if (fighter === 'chong') data.chongScore = Math.max(0, data.chongScore + amount);
                if (fighter === 'hong') data.hongScore = Math.max(0, data.hongScore + amount);
                return data;
            });
        }

        function initJudge() {
            matchRef.on('value', (snapshot) => {

                if (!data) return;
                document.getElementById('judge-round').innerText = data.currentRound;

                    statusEl.className = "text-rose-400 font-bold";

        }

        function handlePointGapWin(data) {
            let winner = data.chongScore > data.hongScore ? 'chong' : 'hong';
            processRoundWinner(winner, `ชนะโดยกฎ Point Gap ห่างกัน 15 แต้ม (${data.chongScore} - ${data.hongScore})`);
        }

        function handleRoundEnd(data) {
            let chong = data.chongScore;
            let hong = data.hongScore;
            let winner = null;
            let tieReason = "";

            if (chong > hong) { winner = 'chong'; tieReason = `คะแนนมากกว่าในยกนี้ (${chong} ต่อ ${hong})`; }
            else if (hong > chong) { winner = 'hong'; tieReason = `คะแนนมากกว่าในยกนี้ (${hong} ต่อ ${chong})`; }
            else {
                if (data.chongSpinHead > data.hongSpinHead) { winner = 'chong'; tieReason = `คะแนนเสมอกัน แต่น้ำเงินได้แต้มลูกหมุนหัวมากกว่า`; }
                else if (data.hongSpinHead > data.chongSpinHead) { winner = 'hong'; tieReason = `คะแนนเสมอกัน แต่แดงได้แต้มลูกหมุนหัวมากกว่า`; }
                else {
                    if (data.chongFirstPoint) { winner = 'chong'; tieReason = `น้ำเงินทำแต้มแรกได้ก่อนในยกนี้`; }
                    else if (data.hongFirstPoint) { winner = 'hong'; tieReason = `แดงทำแต้มแรกได้ก่อนในยกนี้`; }
                    else { showSuperiorityModal(); return; }
                }
            }
            processRoundWinner(winner, tieReason);
        }

        function processRoundWinner(winner, reason) {
            matchRef.once('value', (snapshot) => {
                let data = snapshot.val();
                let cWon = data.chongRoundsWon;
                let hWon = data.hongRoundsWon;
                if (winner === 'chong') cWon++; else hWon++;
                
                let isMatchOver = (cWon >= 2 || hWon >= 2);
                matchRef.update({ chongRoundsWon: cWon, hongRoundsWon: hWon, matchOver: isMatchOver });

                let fighterName = winner === 'chong' ? data.chongName : data.hongName;
                let sideLabel = winner === 'chong' ? 'ฝ่ายน้ำเงิน (CHONG)' : 'ฝ่ายแดง (HONG)';

                let championBadge = document.getElementById('match-champion-badge');
                let championBigBox = document.getElementById('champion-name-big');
                let btnNext = document.getElementById('btn-next-round');

                if (isMatchOver) {
                    championBadge.classList.remove('hidden');
                    document.getElementById('winner-title').innerText = `👑 GRAND CHAMPION 👑`;
                    championBigBox.classList.remove('hidden');
                    championBigBox.innerText = `${fighterName}`;
                    championBigBox.className = `text-4xl md:text-6xl font-black font-cyber py-6 px-8 rounded-2xl border-2 my-2 ${winner === 'chong' ? 'bg-blue-950/90 border-blue-400 text-blue-300 glow-blue' : 'bg-red-950/90 border-red-400 text-red-300 glow-red'}`;
                    document.getElementById('winner-desc').innerText = `สุดยอดนักกีฬา ${sideLabel} คว้าชัยชนะการแข่งขันรายการนี้ด้วยสกอร์ ${cWon} ต่อ ${hWon} ยกอย่างยิ่งใหญ่!`;
                    btnNext.innerText = "🔄 เริ่มแมตช์ใหม่";

                    saveMatchHistory({
                        matchTitle: data.matchTitle,
                        chongName: data.chongName,
                        hongName: data.hongName,
                        score: `${cWon} - ${hWon} ยก`,
                        winner: fighterName,
                        timestamp: new Date().toLocaleString('th-TH')
                    });
                } else {
                    championBadge.classList.add('hidden');
                    championBigBox.classList.add('hidden');
                    document.getElementById('winner-title').innerText = `✨ ผู้ชนะยกที่ ${data.currentRound}: ${fighterName}`;
                    document.getElementById('winner-desc').innerText = `${sideLabel} คว้ายกนี้ไปได้ (สกอร์รวม: น้ำเงิน ${cWon} - ${hWon} แดง)`;
                    btnNext.innerText = "พักยก (Start Rest) ➔";
                }

                document.getElementById('tiebreak-box').classList.remove('hidden');
                document.getElementById('tiebreak-reason').innerText = reason;
                document.getElementById('superiority-box').classList.add('hidden');
                document.getElementById('winner-modal').classList.remove('hidden');
            });
        }
